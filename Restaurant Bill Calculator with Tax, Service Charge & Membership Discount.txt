# Read input values
food_amount = float(input())
day = input().strip()
membership = input().strip()

# 1. Calculate GST (5% of the base food amount)
gst = 0.05 * food_amount
current_total = food_amount + gst

# 2. Calculate Service Charge (on base food amount) based on the day
if day == "Saturday" or day == "Sunday":
    service_charge = 0.10 * food_amount
else:
    service_charge = 0.05 * food_amount

# Add service charge to the running total
current_total = current_total + service_charge

# 3. Apply Membership Discount on the total accumulated so far
if membership == "Gold":
    discount = 0.08 * current_total
elif membership == "Silver":
    discount = 0.04 * current_total
else:
    discount = 0.0

final_bill = current_total - discount

# 4. Calculate Loyalty Cashback based on the final bill amount
if final_bill > 3000:
    cashback = 0.02 * final_bill
else:
    cashback = 0.0

# Print the results rounded to 2 decimal places
print(f"{final_bill:.2f}")
print(f"{cashback:.2f}")
